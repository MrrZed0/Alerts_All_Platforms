using System;
using Newtonsoft.Json;

// Streamer.bot -> OBS alert broadcaster
// Attach this one C# sub-action to any Twitch, YouTube, Kick, Streamlabs or
// StreamElements trigger. Those need no arguments - the platform and the alert
// type are read from the trigger name.
//
// TIKTOK
// Streamer.bot has no TikTok platform, so TikTok cannot fire a native trigger.
// It has to come through a bridge such as TikFinity, which runs one of your
// Streamer.bot actions directly. That means the event type is generic, so the
// script works out TikTok a different way:
//
//   - Gifts and likes are detected automatically, because TikFinity sends
//     giftName / coins / repeatCount / likeCount and nothing else does.
//   - For anything else (shares, follows, subs), make a small action per event
//     with two Set Argument sub-actions ahead of this one:
//
//         platform   = tiktok
//         alertType  = tiktokshare    (or tiktokgift / tiktoklike / follow / sub)
//
// Setting those two explicitly always wins, on any platform. They are also how
// you would wire up any other third-party tool that runs actions directly.
//
// TikFinity's own field names are read automatically, including its spelling of
// %profilePicturUrl%.
public class CPHInline
{
    // Returns the first non-empty argument from a list of possible names
    string Arg(params string[] names)
    {
        foreach (string n in names)
            if (CPH.TryGetArg(n, out object v) && v != null && !string.IsNullOrWhiteSpace(v.ToString()))
                return v.ToString();
        return "";
    }

    public bool Execute()
    {
        string trigger = CPH.GetEventType().ToString().ToLower();

        // 1. Platform.
        //    An explicit "platform" argument always wins. Set one when the
        //    trigger name cannot say which platform it is - TikFinity and other
        //    third-party tools run an action directly, so the event type is
        //    something generic like "Misc.DoAction".
        string platform = Arg("platform", "alertPlatform").ToLower();
        if (platform.Contains("tiktok") || platform == "tt") platform = "tiktok";
        else if (platform.Contains("youtube") || platform == "yt") platform = "youtube";
        else if (platform.Contains("kick")) platform = "kick";
        else if (platform.Contains("streamelements")) platform = "streamelements";
        else if (platform.Contains("streamlabs")) platform = "streamlabs";
        else if (platform.Contains("twitch")) platform = "twitch";
        else if (trigger.Contains("youtube")) platform = "youtube";
        else if (trigger.Contains("tiktok")) platform = "tiktok";
        else if (trigger.Contains("kick")) platform = "kick";
        else if (trigger.Contains("streamelements")) platform = "streamelements";
        else if (trigger.Contains("streamlabs")) platform = "streamlabs";
        //  Nothing in the trigger name. TikFinity (and other TikTok bridges)
        //  run an action directly, so the event type is generic - but they send
        //  fields no other platform uses. Treat those as TikTok.
        else if (!string.IsNullOrEmpty(Arg("giftName", "giftId", "likeCount", "totalLikeCount", "coins", "diamondCount")))
            platform = "tiktok";
        else platform = "twitch";

        // 2. User + avatar
        string user = Arg("user", "userName", "displayName", "nickname", "donationFrom", "from", "donor", "name");
        if (string.IsNullOrEmpty(user)) user = "Anonymous";

        //    "profilePicturUrl" is TikFinity's own spelling of its placeholder - keep both.
        string avatar = Arg("targetUserProfileImageUrl", "userProfileImageUrl", "profileImageUrl",
                            "profilePicturUrl", "profilePictureUrl", "avatar", "userAvatar");
        if (string.IsNullOrEmpty(avatar) && platform == "twitch" && user != "Anonymous")
        {
            try
            {
                var info = CPH.TwitchGetExtendedUserInfoByLogin(user);
                if (info != null && !string.IsNullOrEmpty(info.ProfileImageUrl)) avatar = info.ProfileImageUrl;
            }
            catch { }
        }

        // 3. Alert type. An explicit "alertType" argument wins, same as platform.
        //    Valid values: follow sub resub giftsub raid cheer tip superchat
        //    supersticker member membergift milestone merch host charity
        //    shoutout tiktokgift tiktoklike tiktokshare
        string type = Arg("alertType", "eventType").ToLower();
        if (string.IsNullOrEmpty(type))
        {
        if (platform == "tiktok" && trigger.Contains("gift")) type = "tiktokgift";
        else if (platform == "tiktok" && trigger.Contains("like")) type = "tiktoklike";
        else if (platform == "tiktok" && trigger.Contains("share")) type = "tiktokshare";
        else if (trigger.Contains("supersticker")) type = "supersticker";
        else if (trigger.Contains("superchat")) type = "superchat";
        else if (trigger.Contains("milestone")) type = "milestone";
        else if (trigger.Contains("giftmembership") || trigger.Contains("membershipgift")) type = "membergift";
        else if (trigger.Contains("sponsor") || trigger.Contains("member")) type = "member";
        else if (trigger.Contains("merch")) type = "merch";
        else if (trigger.Contains("charity")) type = "charity";
        else if (trigger.Contains("donation") || trigger.Contains("tip")) type = "tip";
        else if (platform == "youtube" && trigger.Contains("subscriber")) type = "follow";
        else if (trigger.Contains("raid")) type = "raid";
        else if (trigger.Contains("host")) type = "host";
        else if (trigger.Contains("gift")) type = "giftsub";
        else if (trigger.Contains("resub")) type = "resub";
        else if (trigger.Contains("sub")) type = "sub";
        else if (trigger.Contains("cheer") || trigger.Contains("kicks") || trigger.Contains("bits")) type = "cheer";
        else if (platform == "tiktok" && !string.IsNullOrEmpty(Arg("giftName", "giftId"))) type = "tiktokgift";
        else if (platform == "tiktok" && !string.IsNullOrEmpty(Arg("likeCount", "totalLikeCount"))) type = "tiktoklike";
        else type = "follow";
        }

        // 4. Details (argument names differ per integration - check the trigger's variables in Streamer.bot
        //    and add names to these lists if one of your fields comes through empty)
        string amount    = Arg("formattedAmount", "amount", "bits", "kicks", "donationAmount", "diamondCount", "coins");
        string likes     = Arg("likeCount", "totalLikeCount");
        string currency  = Arg("currency", "currencyCode");
        string message   = Arg("message", "donationMessage", "rawInput", "giftName");
        string months    = Arg("cumulative", "cumulativeMonths", "months", "memberMonths", "subMonth");
        string count     = Arg("total", "giftCount", "gifts", "count", "membershipCount", "repeatCount");
        if (string.IsNullOrEmpty(count)) count = likes;   // likes alerts count likes
        string viewers   = Arg("viewers", "viewerCount");
        string recipient = Arg("recipientUser", "recipientUserName", "recipient");
        if (string.IsNullOrEmpty(viewers)) viewers = "1";
        if (string.IsNullOrEmpty(recipient)) recipient = "the community";

        string title = Arg("alertTitle");
        if (string.IsNullOrEmpty(title) && platform == "youtube" && type == "follow") title = "NEW SUBSCRIBER";

        // 5. Optional per-action overrides (set these as arguments on the action)
        var payload = new
        {
            alert = true,
            platform = platform,
            type = type,
            title = title,
            user = user,
            avatar = avatar,
            detail = "",
            useTemplate = true,
            recipient = recipient,
            viewers = viewers,
            months = months,
            amount = amount,
            currency = currency,
            message = message,
            count = count,
            color = Arg("alertColor"),
            inEffect = Arg("inEffect"),
            outEffect = Arg("outEffect"),
            sound = Arg("alertSound")
        };

        CPH.WebsocketBroadcastJson(JsonConvert.SerializeObject(payload));
        return true;
    }
}
